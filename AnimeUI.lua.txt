    CHANGELOG v3.1
    • Background image and overlay now have explicit UICorner — fixes sharp
      edges clipping outside the rounded main frame when ScaleType=Crop.
    • Persistent settings (ZenithImmortal/settings.json):
        – backgroundId  saved/loaded across sessions
        – notificationsEnabled (default true)
        – miniLogoEnabled (default true)
    • Mini logo: draggable circle button, middle-left of screen, click
      to open/close the main UI.  Exposed in BuildSettingsUI.
    • BuildSettingsUI() method — call with any Tab to add permanent
      Background ID input, Notifications toggle, Mini Logo toggle.
    • Slider/Keybind UserInputService connections are now tracked in
      Window._connections so they are properly disconnected on Unload.
    • Shared slider drag state — a single UIS.InputChanged/InputEnded
      pair handles ALL sliders instead of one per slider.
    • Removed dead wtw wrapper (was defined but never used internally).
    • Notification spam on button clicks removed — only toggles, sliders
      on release, dropdowns, and keybinds send notifications.
    • Window:Notify supports c2.Force=true to bypass notifications toggle.
]]

-- HELPERS  (stateless, module-level)

local T  -- set to self.Theme inside CreateWindow

local function tw(obj, t, props, style)
    local tween = TweenService:Create(obj,
        TweenInfo.new(t, style or Enum.EasingStyle.Quart, Enum.EasingDirection.Out),
        props)
    tween:Play()
    return tween
end

local function corner(obj, r)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, r or 8)
    c.Parent = obj
    return c
end

local function pillCorner(obj)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(1, 0)
    c.Parent = obj
    return c
end

local function stroke(obj, col, thick, trans)
    local s        = Instance.new("UIStroke")
    s.Color        = col   or Color3.fromRGB(45, 38, 75)
    s.Thickness    = thick or 1
    s.Transparency = trans or 0
    s.Parent       = obj
    return s
end

local function pad(obj, t, b, l, r)
    local p         = Instance.new("UIPadding")
    p.PaddingTop    = UDim.new(0, t or 8)
    p.PaddingBottom = UDim.new(0, b or 8)
    p.PaddingLeft   = UDim.new(0, l or 8)
    p.PaddingRight  = UDim.new(0, r or 8)
    p.Parent        = obj
end

local function lbl(parent, text, sz, col, font, xA, zi)
    local l                  = Instance.new("TextLabel")
    l.BackgroundTransparency = 1
    l.Text                   = text  or ""
    l.TextSize               = sz    or 13
    l.TextColor3             = col   or Color3.fromRGB(242, 240, 255)
    l.Font                   = font  or Enum.Font.Gotham
    l.TextXAlignment         = xA    or Enum.TextXAlignment.Left
    l.ZIndex                 = zi    or 8
    l.RichText               = true
    l.Parent                 = parent
    return l
end

-- Filesystem helpers (all pcall-wrapped — safe in Studio)
local function fsWrite(path, data)  pcall(writefile, path, data) end
local function fsRead(path)
    local ok, val = pcall(readfile, path)
    return ok and val or nil
end
local function fsMakeFolder(path)   pcall(makefolder, path) end
local function fsIsFile(path)
    local ok, val = pcall(isfile, path)
    return ok and val
end
local function fsListFiles(path)
    local ok, val = pcall(listfiles, path)
    return ok and val or {}
end
local function fsDelete(path)       pcall(delfile, path) end

------------------------------------------------------------------------
-- PERSISTENT SETTINGS  (loaded once per session at window creation)
------------------------------------------------------------------------
local function loadPersistentSettings()
    local raw = fsRead(SETTINGS_FILE)
    if raw then
        local ok, data = pcall(function() return HttpService:JSONDecode(raw) end)
        if ok and type(data) == "table" then return data end
    end
    return {}
end

local function savePersistentSettings(data)
    fsMakeFolder("ZenithImmortal")
    local ok, encoded = pcall(function() return HttpService:JSONEncode(data) end)
    if ok then fsWrite(SETTINGS_FILE, encoded) end
end

        sharedTip.Size = UDim2.new(0, w, 0, 26)

            if not btnFrame.Parent then return end

            local mainAbs = main.AbsolutePosition
            sharedTip.Position = UDim2.new(0, mainAbs.X + SB_W + 8, 0, mouse.Y - 13)
            sharedTip.Visible  = true

    ------------------------------------------------------------------
    local function computeMainSize()
        local vp  = workspace.CurrentCamera.ViewportSize
        -- Scale down so the window always fits with a small margin
        local scl = math.min(1, math.min(vp.X / (WIN_W + 24), vp.Y / (WIN_H + 24)) * 0.94)
        return UDim2.new(0, math.round(WIN_W * scl), 0, math.round(WIN_H * scl))
    end

    local function centerMain(sz)
        main.Position = UDim2.new(0.5, -sz.X.Offset / 2, 0.5, -sz.Y.Offset / 2)
    end

    -- Apply initial size
    local origSize = computeMainSize()
    main.Size     = origSize
    centerMain(origSize)

    -- Refit on viewport change (device rotation, window resize)
    workspace.CurrentCamera:GetPropertyChangedSignal("ViewportSize"):Connect(function()
        origSize  = computeMainSize()
        if isOpen then
            main.Size = origSize
            centerMain(origSize)
        end
    end)

        isOpen = val
        if isOpen then
            main.Visible = true
            origSize     = computeMainSize()   -- recompute in case viewport changed
            main.Size    = UDim2.new(0, math.round(origSize.X.Offset * 0.9),
                                     0, math.round(origSize.Y.Offset * 0.9))
            centerMain(main.Size)
            tw(main, 0.22, {Size = origSize}, Enum.EasingStyle.Back)
        else
            tw(main, 0.17, {Size = UDim2.new(0, math.round(origSize.X.Offset * 0.85),
                                              0, math.round(origSize.Y.Offset * 0.85))})
            task.delay(0.19, function() main.Visible = false end)
        end
    end

    UserInputService.InputBegan:Connect(function(i, gp)
        if gp then return end
        if i.KeyCode == Enum.KeyCode.RightShift then setOpen(not isOpen) end
    end)

    local mlDragOn    = false
    local mlDragInp   = nil
    local mlDragStart = nil
    local mlPosStart  = nil
    local mlMoved     = false

    miniLogo.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then
            mlDragOn    = true
            mlMoved     = false
            mlDragStart = i.Position
            mlPosStart  = miniLogo.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then
                    mlDragOn = false
                end
            end)
        end
    end)
    miniLogo.InputChanged:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseMovement then
            mlDragInp = i
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if i == mlDragInp and mlDragOn and mlDragStart then
            local d = i.Position - mlDragStart
            if d.Magnitude > 4 then mlMoved = true end
            miniLogo.Position = UDim2.new(
                mlPosStart.X.Scale, mlPosStart.X.Offset + d.X,
                mlPosStart.Y.Scale, mlPosStart.Y.Offset + d.Y
            )
        end
    end)

    -- Click only fires if not a drag
    miniLogo.MouseButton1Click:Connect(function()
        if not mlMoved then
            setOpen(not isOpen)
        end
    end)

    ------------------------------------------------------------------
    -- SHARED SLIDER DRAG STATE
    -- A single pair of UIS connections handles ALL sliders instead of
    -- one pair per slider — avoids connection accumulation.
    ------------------------------------------------------------------
    local activeDragFn  = nil  -- set/cleared by individual slider's InputBegan
    local activeDragEnd = nil  -- called on MouseButton1 release

    UserInputService.InputChanged:Connect(function(i)
        if activeDragFn and i.UserInputType == Enum.UserInputType.MouseMovement then
            activeDragFn(i.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 and activeDragEnd then
            activeDragEnd()
            activeDragFn  = nil
            activeDragEnd = nil
        end
    end)

    ------------------------------------------------------------------
    -- UNLOAD
    ------------------------------------------------------------------
    function Window:Unload()
        if not self._alive then return end
        self._alive = false
        self:Notify({ Title="Zenith Immortal", Content="UI Unloaded.", Type="warn", Duration=2, Force=true })
        task.delay(0.5, function()
            for _, c in ipairs(self._connections) do pcall(function() c:Disconnect() end) end
            for _, t in ipairs(self._tasks)       do pcall(function() task.cancel(t) end) end
            pcall(function() gui:Destroy() end)
        end)
    end

    -- CONFIG SYSTEM
    ------------------------------------------------------------------
    local function ensureFolder()
        fsMakeFolder("ZenithImmortal")
        fsMakeFolder(CONFIG_FOLDER)
    end

    function Window:SaveConfig(name)
        if not name or name == "" then
            self:Notify({ Title="Config", Content="Name is empty.", Type="error", Duration=3 })
            return
        end
        ensureFolder()
        local data = {}
        for flag, obj in pairs(self.Flags) do data[flag] = obj.Get() end
        local ok, encoded = pcall(function() return HttpService:JSONEncode(data) end)
        if ok then
            fsWrite(CONFIG_FOLDER .. "/" .. name .. ".json", encoded)
            self:Notify({ Title="Config Saved", Content=name, Type="success", Duration=3 })
        else
            self:Notify({ Title="Config Error", Content="Failed to encode.", Type="error", Duration=3 })
        end
    end

    function Window:LoadConfig(name)
        local path = CONFIG_FOLDER .. "/" .. name .. ".json"
        if not fsIsFile(path) then
            self:Notify({ Title="Config Error", Content=name.." not found.", Type="error", Duration=3 })
            return
        end
        local raw = fsRead(path)
        if not raw then
            self:Notify({ Title="Config Error", Content="Could not read file.", Type="error", Duration=3 })
            return
        end
        local ok, data = pcall(function() return HttpService:JSONDecode(raw) end)
        if not ok then
            self:Notify({ Title="Config Error", Content="Failed to parse.", Type="error", Duration=3 })
            return
        end
        for flag, value in pairs(data) do
            if self.Flags[flag] then pcall(function() self.Flags[flag].Set(value) end) end
        end
        self:Notify({ Title="Config Loaded", Content=name, Type="success", Duration=3 })
    end

    function Window:DeleteConfig(name)
        local path = CONFIG_FOLDER .. "/" .. name .. ".json"
        if fsIsFile(path) then
            fsDelete(path)
            self:Notify({ Title="Config Deleted", Content=name, Type="warn", Duration=3 })
        else
            self:Notify({ Title="Config Error", Content=name.." not found.", Type="error", Duration=3 })
        end
    end

    function Window:ListConfigs()
        ensureFolder()
        local files = fsListFiles(CONFIG_FOLDER)
        local names = {}
        for _, f in ipairs(files) do
            local name = f:match("([^/\\]+)%.json$")
            if name then table.insert(names, name) end
        end
        return names
    end

    function Window:ExportConfig(name)
        local path = CONFIG_FOLDER .. "/" .. name .. ".json"
        if fsIsFile(path) then
            pcall(setclipboard, fsRead(path))
            self:Notify({ Title="Config Exported", Content="Copied to clipboard.", Type="success", Duration=3 })
        else
            self:Notify({ Title="Export Error", Content=name.." not found.", Type="error", Duration=3 })
        end
    end

    function Window:ImportConfig(json, name)
        if not json or json == "" then
            self:Notify({ Title="Import Error", Content="No data provided.", Type="error", Duration=3 })
            return
        end
        ensureFolder()
        fsWrite(CONFIG_FOLDER .. "/" .. (name or "imported") .. ".json", json)
        self:Notify({ Title="Config Imported", Content=name or "imported", Type="success", Duration=3 })
    end

            local value = math.clamp(c2.Default or minV, minV, maxV)

            local box                  = Instance.new("Frame")
            box.Size                   = UDim2.new(1, 0, 0, 54)
            box.BackgroundColor3       = T.Card
            box.LayoutOrder            = nxt()
            box.ZIndex                 = Z.Controls
            box.Parent                 = page
            corner(box, 10)
            stroke(box, T.Stroke, 1)

            local nm = lbl(box, c2.Name or "Slider", 13,
                T.Text, Enum.Font.GothamSemibold, Enum.TextXAlignment.Left, Z.Controls+1)
            nm.Size     = UDim2.new(1, -70, 0, 20)
            nm.Position = UDim2.new(0, 12, 0, 8)

            local vLbl = lbl(box, tostring(value)..suf, 12,
                T.Accent, Enum.Font.GothamBold, Enum.TextXAlignment.Right, Z.Controls+1)
            vLbl.Size     = UDim2.new(0, 56, 0, 20)
            vLbl.Position = UDim2.new(1, -66, 0, 8)

            local track                = Instance.new("Frame")
            track.Size                 = UDim2.new(1, -24, 0, 6)
            track.Position             = UDim2.new(0, 12, 0, 38)
            track.BackgroundColor3     = T.CardHover
            track.ZIndex               = Z.Controls + 1
            track.Parent               = box
            pillCorner(track)

            local fill                 = Instance.new("Frame")
            fill.Size                  = UDim2.new((value-minV)/(maxV-minV), 0, 1, 0)
            fill.BackgroundColor3      = T.Accent
            fill.ZIndex                = Z.Controls + 2
            fill.Parent                = track
            pillCorner(fill)

            local fg = Instance.new("UIGradient")
            fg.Color  = ColorSequence.new(T.AccentDim, T.AccentGlow)
            fg.Parent = fill

            local thumb                = Instance.new("Frame")
            thumb.Size                 = UDim2.new(0, 14, 0, 14)
            thumb.AnchorPoint          = Vector2.new(0.5, 0.5)
            thumb.Position             = UDim2.new((value-minV)/(maxV-minV), 0, 0.5, 0)
            thumb.BackgroundColor3     = Color3.new(1, 1, 1)
            thumb.ZIndex               = Z.Controls + 3
            thumb.Parent               = track
            corner(thumb, 7)
            local tS = stroke(thumb, T.Accent, 2)

            local function glowThumb(on)
                tw(thumb, 0.13, {Size = on
                    and UDim2.new(0, 18, 0, 18) or UDim2.new(0, 14, 0, 14)})
                tw(tS, 0.13, {Color = on and T.AccentGlow or T.Accent})
            end

            local function setV(x)
                local p = math.clamp(
                    (x - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
                value = math.floor(minV + p * (maxV - minV) + 0.5)
                local pn = (value - minV) / (maxV - minV)
                tw(fill,  0.04, {Size     = UDim2.new(pn, 0, 1, 0)})
                tw(thumb, 0.04, {Position = UDim2.new(pn, 0, 0.5, 0)})
                vLbl.Text = tostring(value) .. suf
                if c2.Callback then c2.Callback(value) end
            end

            -- Use the SHARED drag state instead of per-slider UIS connections
            track.InputBegan:Connect(function(i)
                if i.UserInputType == Enum.UserInputType.MouseButton1 then
                    glowThumb(true)
                    setV(i.Position.X)
                    activeDragFn  = function(x) setV(x) end
                    activeDragEnd = function()
                        glowThumb(false)
                        if not Window._alive then return end
                        Window:Notify({
                            Title    = c2.Name or "Slider",
                            Content  = "Set to " .. tostring(value) .. suf,
                            Duration = 2,
                        })
                    end
                end
            end)

            local Sl = {}
            function Sl:Set(v)
                value = math.clamp(v, minV, maxV)
                local pn = (value-minV)/(maxV-minV)
                tw(fill,  0.14, {Size     = UDim2.new(pn, 0, 1, 0)})
                tw(thumb, 0.14, {Position = UDim2.new(pn, 0, 0.5, 0)})
                vLbl.Text = tostring(value)..suf
                if c2.Callback then c2.Callback(value) end
            end
            function Sl:Get() return value end

            if c2.Flag then
                Window.Flags[c2.Flag] = { Type="Slider", Set=Sl.Set, Get=Sl.Get }
            end
            return Sl
        end

        ---- DROPDOWN ----
        function Tab:AddDropdown(c2)
            c2 = c2 or {}
            local opts  = c2.Options or {}
            local sel   = c2.Default or (opts[1] or "None")
            local open2 = false

            local box                  = Instance.new("Frame")
            box.Size                   = UDim2.new(1, 0, 0, 42)
            box.BackgroundColor3       = T.Card
            box.LayoutOrder            = nxt()
            box.ClipsDescendants       = false
            box.ZIndex                 = Z.Controls
            box.Parent                 = page
            corner(box, 10)
            local dS = stroke(box, T.Stroke, 1)

            local nm = lbl(box, c2.Name or "Dropdown", 13,
                T.Text, Enum.Font.GothamSemibold, Enum.TextXAlignment.Left, Z.Controls+1)
            nm.Size     = UDim2.new(0.46, 0, 1, 0)
            nm.Position = UDim2.new(0, 12, 0, 0)

            local selB                 = Instance.new("TextButton")
            selB.Size                  = UDim2.new(0.48, 0, 0, 28)
            selB.Position              = UDim2.new(0.5, 0, 0.5, -14)
            selB.BackgroundColor3      = T.CardHover
            selB.Text                  = sel
            selB.TextColor3            = T.Accent
            selB.Font                  = Enum.Font.GothamSemibold
            selB.TextSize              = 12
            selB.AutoButtonColor       = false
            selB.ZIndex                = Z.Controls + 1
            selB.Parent                = box
            corner(selB, 7)
            pad(selB, 0, 0, 8, 22)

            local chev = lbl(selB, "▾", 11, T.TextMuted,
                Enum.Font.GothamBold, Enum.TextXAlignment.Right, Z.Controls+2)
            chev.Size  = UDim2.new(1, -4, 1, 0)

            -- Dropdown panel on overlay (always above controls)
            local dp                          = Instance.new("ScrollingFrame")
            dp.Name                           = "DropPanel_" .. (c2.Name or "")
            dp.Size                           = UDim2.new(0, 0, 0, 0)
            dp.BackgroundColor3               = T.Panel
            dp.ClipsDescendants               = true
            dp.Active                         = true
            dp.ZIndex                         = Z.Dropdown
            dp.Visible                        = false
            dp.ScrollBarThickness             = 3
            dp.ScrollBarImageColor3           = T.Accent
            dp.ScrollBarImageTransparency     = 0.4
            dp.CanvasSize                     = UDim2.new(0, 0, 0, 0)
            dp.AutomaticCanvasSize            = Enum.AutomaticSize.Y
            dp.ScrollingDirection             = Enum.ScrollingDirection.Y
            dp.Parent                         = dropOverlay
            corner(dp, 8)
            stroke(dp, T.Accent, 1)
            pad(dp, 4, 4, 4, 4)

            local dpL         = Instance.new("UIListLayout")
            dpL.Padding       = UDim.new(0, 2)
            dpL.SortOrder     = Enum.SortOrder.LayoutOrder
            dpL.Parent        = dp

            local function closeD()
                if not open2 then return end
                open2 = false
                if activeDropClose == closeD then activeDropClose = nil end
                tw(dp, 0.13, {Size = UDim2.new(0, dp.AbsoluteSize.X, 0, 0)})
                task.delay(0.14, function() dp.Visible = false end)
                tw(chev, 0.13, {Rotation = 0})
                tw(dS,   0.13, {Color = T.Stroke})
            end

            local function buildOpts()
                for _, ch in pairs(dp:GetChildren()) do
                    if ch:IsA("TextButton") then ch:Destroy() end
                end
                for i, opt in ipairs(opts) do
                    local ob                 = Instance.new("TextButton")
                    ob.Size                  = UDim2.new(1, 0, 0, 26)
                    ob.BackgroundColor3      = T.Card
                    ob.Text                  = opt
                    ob.TextColor3            = opt == sel and T.Accent or T.Text
                    ob.Font                  = Enum.Font.Gotham
                    ob.TextSize              = 11
                    ob.AutoButtonColor       = false
                    ob.LayoutOrder           = i
                    ob.ZIndex                = Z.Dropdown + 1
                    ob.Parent                = dp
                    corner(ob, 5)
                    ob.MouseEnter:Connect(function()
                        tw(ob, 0.09, {BackgroundColor3 = T.CardHover, TextColor3 = T.AccentGlow})
                    end)
                    ob.MouseLeave:Connect(function()
                        tw(ob, 0.09, {BackgroundColor3 = T.Card,
                            TextColor3 = opt == sel and T.Accent or T.Text})
                    end)
                    ob.MouseButton1Click:Connect(function()
                        if not Window._alive then return end
                        sel = opt; selB.Text = opt
                        closeD(); buildOpts()
                        Window:Notify({
                            Title    = c2.Name or "Dropdown",
                            Content  = "Selected: " .. opt,
                            Duration = 2,
                        })
                        if c2.Callback then c2.Callback(opt) end
                    end)
                end
            end
            buildOpts()

            selB.MouseButton1Click:Connect(function()
                if not Window._alive then return end
                if open2 then closeD(); return end

                -- Support Refresh callback (from user's gsub patch)
                if c2.Refresh then
                    local refreshed = c2.Refresh()
                    if refreshed then opts = refreshed; buildOpts() end
                end

                if activeDropClose and activeDropClose ~= closeD then
                    activeDropClose()
                end
                open2 = true
                activeDropClose = closeD

                local absPos  = selB.AbsolutePosition
                local absSize = selB.AbsoluteSize
                local panelW  = absSize.X + 6   -- +6 so scrollbar doesn't overlap options
                local panelH  = math.min(#opts * 30 + 10, 200)  -- 200px visible, scrolls beyond

                local screenH    = workspace.CurrentCamera.ViewportSize.Y
                local openUp     = (screenH - (absPos.Y + absSize.Y)) < panelH + 10

                dp.Size     = UDim2.new(0, panelW, 0, 0)
                dp.CanvasPosition = Vector2.zero  -- scroll back to top on reopen
                dp.Position = openUp
                    and UDim2.new(0, absPos.X, 0, absPos.Y - panelH - 4)
                    or  UDim2.new(0, absPos.X, 0, absPos.Y + absSize.Y + 4)
                dp.Visible = true

                tw(dp,   0.17, {Size = UDim2.new(0, panelW, 0, panelH)})
                tw(chev, 0.17, {Rotation = 180})
                tw(dS,   0.17, {Color = T.Accent})
            end)

            local DD = {}
            function DD:Set(v) sel=v; selB.Text=v; buildOpts() end
            function DD:Get() return sel end
            function DD:SetOptions(o) opts=o; buildOpts() end

            if c2.Flag then
                Window.Flags[c2.Flag] = { Type="Dropdown", Set=DD.Set, Get=DD.Get }
            end
            return DD
        end

                    lstn = false; bind = i.KeyCode
                    kb.Text = bind.Name; kb.TextColor3 = T.Accent
                    if not Window._alive then return end
                    Window:Notify({
                        Title    = c2.Name or "Keybind",
                        Content  = "Bound to " .. bind.Name,
                        Duration = 2,
                    })
                    if c2.Callback then c2.Callback(bind) end
                end
            end)

            local KB = {}
            function KB:Get() return bind end
            function KB:Set(k) bind=k; kb.Text=k.Name end

            if c2.Flag then
                Window.Flags[c2.Flag] = { Type="Keybind", Set=KB.Set, Get=KB.Get }
            end
