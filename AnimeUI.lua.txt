local T = nil  -- set inside CreateWindow from self.Theme

local function tw(obj, t, props, style)
    TweenService:Create(obj,
        TweenInfo.new(t, style or Enum.EasingStyle.Quart, Enum.EasingDirection.Out),
        props):Play()
end

local function corner(obj, r)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, r or 8)
    c.Parent = obj
    return c
end

local function mkStroke(obj, col, thick, trans)
    local s = Instance.new("UIStroke")
    s.Color        = col   or Color3.fromRGB(45,38,75)
    s.Thickness    = thick or 1
    s.Transparency = trans or 0
    s.Parent       = obj
    return s
end

local function mkPad(obj, top, bot, left, right)
    local p         = Instance.new("UIPadding")
    p.PaddingTop    = UDim.new(0, top   or 8)
    p.PaddingBottom = UDim.new(0, bot   or 8)
    p.PaddingLeft   = UDim.new(0, left  or 8)
    p.PaddingRight  = UDim.new(0, right or 8)
    p.Parent        = obj
    return p
end

local function mkLabel(parent, text, sz, col, font, xAlign, zi)
    local l                  = Instance.new("TextLabel")
    l.BackgroundTransparency = 1
    l.Text                   = text   or ""
    l.TextSize               = sz     or 13
    l.TextColor3             = col    or Color3.fromRGB(242,240,255)
    l.Font                   = font   or Enum.Font.Gotham
    l.TextXAlignment         = xAlign or Enum.TextXAlignment.Left
    l.ZIndex                 = zi     or 8
    l.RichText               = true
    l.Parent                 = parent
    return l
end

local function draggable(frame, handle)
    local drag, di, mPos, fPos = false, nil, nil, nil
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then
            drag = true; mPos = i.Position; fPos = frame.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then drag = false end
            end)
        end
    end)
    handle.InputChanged:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseMovement then di = i end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if i == di and drag then
            local d = i.Position - mPos
            frame.Position = UDim2.new(fPos.X.Scale, fPos.X.Offset + d.X,
                                        fPos.Y.Scale, fPos.Y.Offset + d.Y)
        end
    end)
end

        -- ACTIVATE this tab
        --------------------------------------------------------
        local function activate()
            for _, p in pairs(contentArea:GetChildren()) do
                if p:IsA("ScrollingFrame") then p.Visible = false end
            end
            for _, b in pairs(sidebar:GetChildren()) do
                if b:IsA("ImageButton") then
                    tw(b, 0.18, {BackgroundColor3 = T.Card, ImageTransparency = 0.25})
                    local s = b:FindFirstChildOfClass("UIStroke")
                    if s then tw(s, 0.18, {Color = T.Stroke}) end
                end
            end
            page.Visible = true
            tw(tabBtn, 0.2, {BackgroundColor3 = T.AccentDim, ImageTransparency = 0})
            tw(tabStroke, 0.2, {Color = T.Accent})
            Window._activeTab = name
        end

        tabBtn.MouseButton1Click:Connect(activate)
        tabBtn.MouseEnter:Connect(function()
            tip.Visible = true
            if Window._activeTab ~= name then
                tw(tabBtn, 0.12, {BackgroundColor3 = T.CardHover, ImageTransparency = 0.1})
            end
        end)
        tabBtn.MouseLeave:Connect(function()
            tip.Visible = false
            if Window._activeTab ~= name then
                tw(tabBtn, 0.12, {BackgroundColor3 = T.Card, ImageTransparency = 0.25})
            end
        end)

        if idx == 1 then task.defer(activate) end

                tw(track, 0.22, {BackgroundColor3 = state and T.Accent or T.CardHover})

                    and UDim2.new(1, -20, 0.5, -9)

            function Toggle:Set(v) state = v; refresh(); if cfg2.Callback then cfg2.Callback(state) end end
            function Toggle:Get() return state end

        end

        --------------------------------------------------------
        -- SLIDER  (thin minimalist track, glowy thumb)
        --------------------------------------------------------
        function Tab:AddSlider(cfg2)
            cfg2 = cfg2 or {}
            local minV  = cfg2.Min    or 0
            local maxV  = cfg2.Max    or 100
            local suf   = cfg2.Suffix or ""
            local value = math.clamp(cfg2.Default or minV, minV, maxV)

            local box                   = Instance.new("Frame")
            box.Size                    = UDim2.new(1, 0, 0, 54)
            box.BackgroundColor3        = T.Card
            box.LayoutOrder             = nextOrder()
            box.ZIndex                  = 12
            box.Parent                  = page
            corner(box, 10)
            mkStroke(box, T.Stroke, 1)

            -- Name
            local sName = mkLabel(box, cfg2.Name or "Slider",
                13, T.Text, Enum.Font.GothamSemibold, Enum.TextXAlignment.Left, 13)
            sName.Size     = UDim2.new(1, -70, 0, 20)
            sName.Position = UDim2.new(0, 12, 0, 8)

            -- Value readout (accent colour, right-aligned)
            local valLbl = mkLabel(box, tostring(value) .. suf,
                12, T.Accent, Enum.Font.GothamBold, Enum.TextXAlignment.Right, 13)
            valLbl.Size     = UDim2.new(0, 56, 0, 20)
            valLbl.Position = UDim2.new(1, -66, 0, 8)

            -- Track (thin — 4px)
            local track                 = Instance.new("Frame")
            track.Size                  = UDim2.new(1, -24, 0, 4)
            track.Position              = UDim2.new(0, 12, 0, 40)
            track.BackgroundColor3      = T.CardHover
            track.ZIndex                = 13
            track.Parent                = box
            corner(track, 2)

            -- Fill with gradient
            local fill                  = Instance.new("Frame")
            fill.Size                   = UDim2.new((value-minV)/(maxV-minV), 0, 1, 0)
            fill.BackgroundColor3       = T.Accent
            fill.ZIndex                 = 14
            fill.Parent                 = track
            corner(fill, 2)

            local fillGrad      = Instance.new("UIGradient")
            fillGrad.Color      = ColorSequence.new(T.AccentDim, T.AccentGlow)
            fillGrad.Parent     = fill

            -- Glowy thumb (slightly larger, white with purple stroke)
            local thumb                 = Instance.new("Frame")
            thumb.Size                  = UDim2.new(0, 14, 0, 14)
            thumb.AnchorPoint           = Vector2.new(0.5, 0.5)
            thumb.Position              = UDim2.new((value-minV)/(maxV-minV), 0, 0.5, 0)
            thumb.BackgroundColor3      = Color3.fromRGB(255, 255, 255)
            thumb.ZIndex                = 15
            thumb.Parent                = track
            corner(thumb, 7)
            local thumbStroke = mkStroke(thumb, T.Accent, 2)

            -- Thumb expands on drag
            local function thumbHover(on)
                tw(thumb, 0.14, {Size = on
                    and UDim2.new(0, 18, 0, 18)
                    or  UDim2.new(0, 14, 0, 14)})
                tw(thumbStroke, 0.14, {Color = on and T.AccentGlow or T.Accent})
            end

            local isDragging = false

            local function setVal(x)
                local abs = track.AbsolutePosition.X
                local w   = track.AbsoluteSize.X
                local pct = math.clamp((x - abs) / w, 0, 1)
                value     = math.floor(minV + pct * (maxV - minV) + 0.5)
                local p   = (value - minV) / (maxV - minV)
                tw(fill,  0.04, {Size     = UDim2.new(p, 0, 1, 0)})
                tw(thumb, 0.04, {Position = UDim2.new(p, 0, 0.5, 0)})
                valLbl.Text = tostring(value) .. suf
                if cfg2.Callback then cfg2.Callback(value) end
            end

            track.InputBegan:Connect(function(inp)
                if inp.UserInputType == Enum.UserInputType.MouseButton1 then
                    isDragging = true
                    thumbHover(true)
                    setVal(inp.Position.X)
                end
            end)
            UserInputService.InputChanged:Connect(function(inp)
                if isDragging and inp.UserInputType == Enum.UserInputType.MouseMovement then
                    setVal(inp.Position.X)
                end
            end)
            UserInputService.InputEnded:Connect(function(inp)
                if inp.UserInputType == Enum.UserInputType.MouseButton1 then
                    isDragging = false
                    thumbHover(false)
                end
            end)

            local Slider = {}
            function Slider:Set(v)
                value = math.clamp(v, minV, maxV)
                local p = (value-minV)/(maxV-minV)
                tw(fill,  0.14, {Size     = UDim2.new(p, 0, 1, 0)})
                tw(thumb, 0.14, {Position = UDim2.new(p, 0, 0.5, 0)})
                valLbl.Text = tostring(value)..suf
                if cfg2.Callback then cfg2.Callback(value) end
            end
            function Slider:Get() return value end
            return Slider
        end

        function Tab:AddDropdown(cfg2)
            cfg2 = cfg2 or {}

            local selected = cfg2.Default or (opts[1] or "None")
            local dropOpen = false

            local box                   = Instance.new("Frame")

            dName.Size     = UDim2.new(0.46, 0, 1, 0)
            dName.Position = UDim2.new(0, 12, 0, 0)

            local selBtn                    = Instance.new("TextButton")
            selBtn.Size                     = UDim2.new(0.48, 0, 0, 28)
            selBtn.Position                 = UDim2.new(0.5, 0, 0.5, -14)
            selBtn.BackgroundColor3         = T.CardHover
            selBtn.Text                     = selected
            selBtn.TextColor3               = T.Accent
            selBtn.Font                     = Enum.Font.GothamSemibold
            selBtn.TextSize                 = 12
            selBtn.AutoButtonColor          = false
            selBtn.ZIndex                   = 13
            selBtn.Parent                   = box
            corner(selBtn, 7)
            mkPad(selBtn, 0, 0, 8, 22)

            local chev = mkLabel(selBtn, "▾", 11, T.TextMuted,
                Enum.Font.GothamBold, Enum.TextXAlignment.Right, 14)
            chev.Size  = UDim2.new(1, -4, 1, 0)

            local dp                    = Instance.new("Frame")
            dp.Size                     = UDim2.new(0.48, 0, 0, 0)
            dp.Position                 = UDim2.new(0.5, 0, 1, 4)
            dp.BackgroundColor3         = T.Panel
            dp.ClipsDescendants         = true
            dp.ZIndex                   = 30
            dp.Visible                  = false
            dp.Parent                   = box
            corner(dp, 8)
            mkStroke(dp, T.Accent, 1)
            mkPad(dp, 4, 4, 4, 4)

            local dpLayout      = Instance.new("UIListLayout")
            dpLayout.Padding    = UDim.new(0, 2)
            dpLayout.SortOrder  = Enum.SortOrder.LayoutOrder
            dpLayout.Parent     = dp

            local function closeD()
                dropOpen = false
                tw(dp, 0.14, {Size = UDim2.new(0.48, 0, 0, 0)})
                task.delay(0.15, function() dp.Visible = false end)
                tw(chev, 0.14, {Rotation = 0})
                tw(dStroke, 0.14, {Color = T.Stroke})
            end

            local function buildOpts()
                for _, c in pairs(dp:GetChildren()) do
                    if c:IsA("TextButton") then c:Destroy() end
                end
                for i, opt in ipairs(opts) do
                    local ob                    = Instance.new("TextButton")
                    ob.Size                     = UDim2.new(1, 0, 0, 26)
                    ob.BackgroundColor3         = T.Card
                    ob.Text                     = opt
                    ob.TextColor3               = opt == selected and T.Accent or T.Text
                    ob.Font                     = Enum.Font.Gotham
                    ob.TextSize                 = 11
                    ob.AutoButtonColor          = false
                    ob.LayoutOrder              = i
                    ob.ZIndex                   = 31
                    ob.Parent                   = dp
                    corner(ob, 5)
                    ob.MouseEnter:Connect(function()
                        tw(ob, 0.1, {BackgroundColor3 = T.CardHover, TextColor3 = T.AccentGlow})
                    end)
                    ob.MouseLeave:Connect(function()
                        tw(ob, 0.1, {BackgroundColor3 = T.Card,
                            TextColor3 = opt == selected and T.Accent or T.Text})
                    end)
                    ob.MouseButton1Click:Connect(function()
                        selected = opt; selBtn.Text = opt
                        closeD(); buildOpts()
                        if cfg2.Callback then cfg2.Callback(opt) end
                    end)
                end
            end
            buildOpts()

            selBtn.MouseButton1Click:Connect(function()
                dropOpen = not dropOpen
                if dropOpen then
                    dp.Visible = true
                    local h = math.min(#opts * 28 + 10, 160)
                    tw(dp, 0.18, {Size = UDim2.new(0.48, 0, 0, h)})
                    tw(chev, 0.18, {Rotation = 180})
                    tw(dStroke, 0.18, {Color = T.Accent})
                else
                    closeD()
                end
            end)

            local DD = {}
            function DD:Set(v) selected = v; selBtn.Text = v; buildOpts() end
            function DD:Get() return selected end
            function DD:SetOptions(o) opts = o; buildOpts() end
            return DD
        end
