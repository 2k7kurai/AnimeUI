local T = ZenithImmortal.Theme

local function tw(obj, t, props, style)
    TweenService:Create(obj, TweenInfo.new(t,
        style or Enum.EasingStyle.Quart, Enum.EasingDirection.Out), props
    ):Play()
end

local function corner(obj, r)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, r or 8)
    c.Parent = obj
    return c
end

local function mkStroke(obj, col, thick, trans)
    local s = Instance.new("UIStroke")
    s.Color = col or T.Stroke
    s.Thickness = thick or 1
    s.Transparency = trans or 0
    s.Parent = obj
    return s
end

local function mkPad(obj, t, b, l, r)
    local p = Instance.new("UIPadding")
    p.PaddingTop    = UDim.new(0, t or 8)
    p.PaddingBottom = UDim.new(0, b or 8)
    p.PaddingLeft   = UDim.new(0, l or 8)
    p.PaddingRight  = UDim.new(0, r or 8)
    p.Parent = obj
    return p
end

local function lbl(parent, text, sz, col, font, xAlign, zIndex)
    local l = Instance.new("TextLabel")
    l.BackgroundTransparency = 1
    l.Text = text or ""
    l.TextSize = sz or 13
    l.TextColor3 = col or T.Text
    l.Font = font or Enum.Font.Gotham
    l.TextXAlignment = xAlign or Enum.TextXAlignment.Left
    l.ZIndex = zIndex or 8
    l.RichText = true
    l.Parent = parent
    return l
end

local function draggable(frame, handle)
    local drag, dragInp, mPos, fPos = false, nil, nil, nil
    handle.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1 then
            drag  = true
            mPos  = inp.Position
            fPos  = frame.Position
            inp.Changed:Connect(function()
                if inp.UserInputState == Enum.UserInputState.End then drag = false end
            end)
        end
    end)
    handle.InputChanged:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseMovement then dragInp = inp end
    end)
    UserInputService.InputChanged:Connect(function(inp)
        if inp == dragInp and drag then
            local d = inp.Position - mPos
            frame.Position = UDim2.new(
                fPos.X.Scale, fPos.X.Offset + d.X,
                fPos.Y.Scale, fPos.Y.Offset + d.Y
            )
        end
    end)
end

        tabBtn.ZIndex           = 6
        tabBtn.Parent           = sidebar
        corner(tabBtn, 10)
        local tabStroke = mkStroke(tabBtn, theme.Stroke, 1.2)

        -- Tooltip
        local tip = Instance.new("Frame")
        tip.Size                 = UDim2.new(0, 0, 0, 26)
        tip.Position             = UDim2.new(1, 10, 0.5, -13)
        tip.BackgroundColor3     = theme.Card
        tip.ClipsDescendants     = true
        tip.Visible              = false
        tip.ZIndex               = 30
        tip.Parent               = tabBtn
        corner(tip, 6)
        mkStroke(tip, theme.StrokeHot, 1)

        local tipLbl = lbl(tip, name, 11, theme.Text, Enum.Font.GothamSemibold,
            Enum.TextXAlignment.Center, 31)
        tipLbl.Size = UDim2.new(1, 0, 1, 0)
        mkPad(tip, 0, 0, 8, 8)

        -- measure text width
        task.defer(function()
            local ts = game:GetService("TextService")
            local w = ts:GetTextSize(name, 11, Enum.Font.GothamSemibold,
                Vector2.new(999,26)).X + 18
            tip.Size = UDim2.new(0, w, 0, 26)
        end)

        -- ── Page ─────────────────────────────────────────────
        local page = Instance.new("ScrollingFrame")
        page.Name                   = "Page_"..name
        page.Size                   = UDim2.new(1, 0, 1, 0)
        page.BackgroundTransparency = 1
        page.Visible                = false
        page.ScrollBarThickness     = 2
        page.ScrollBarImageColor3   = theme.Accent
        page.ScrollBarImageTransparency = 0.5
        page.CanvasSize             = UDim2.new(0,0,0,0)
        page.AutomaticCanvasSize    = Enum.AutomaticSize.Y
        page.ZIndex                 = 6
        page.Parent                 = contentArea
        mkPad(page, 10, 12, 14, 14)

        local pageLayout = Instance.new("UIListLayout")
        pageLayout.Padding   = UDim.new(0, 7)
        pageLayout.SortOrder = Enum.SortOrder.LayoutOrder
        pageLayout.Parent    = page

        -- ── Activate ─────────────────────────────────────────
        local function activate()
            -- deselect all
            for _, p in pairs(contentArea:GetChildren()) do
                if p:IsA("ScrollingFrame") then p.Visible = false end
            end
            for _, b in pairs(sidebar:GetChildren()) do
                if b:IsA("ImageButton") then
                    tw(b, 0.18, {BackgroundColor3=theme.Card, ImageColor3=theme.TextMuted})
                    local s = b:FindFirstChildOfClass("UIStroke")
                    if s then tw(s, 0.18, {Color=theme.Stroke}) end
                end
            end
            -- select this
            page.Visible = true
            tw(tabBtn, 0.2, {BackgroundColor3=theme.AccentDim, ImageColor3=theme.AccentGlow})
            tw(tabStroke, 0.2, {Color=theme.Accent})
            Window._activeTab = name
        end

        tabBtn.MouseButton1Click:Connect(activate)

        tabBtn.MouseEnter:Connect(function()
            tip.Visible = true
            if Window._activeTab ~= name then
                tw(tabBtn, 0.12, {BackgroundColor3=theme.CardHover})
            end
        end)
        tabBtn.MouseLeave:Connect(function()
            tip.Visible = false
            if Window._activeTab ~= name then
                tw(tabBtn, 0.12, {BackgroundColor3=theme.Card})
            end
        end)

        if idx == 1 then task.defer(activate) end

            local rName = lbl(row, cfg2.Name or "Toggle", 13,
                theme.Text, Enum.Font.GothamSemibold, Enum.TextXAlignment.Left, 8)

            if cfg2.Description then
                rName.Size     = UDim2.new(1, -72, 0, 20)
                rName.Position = UDim2.new(0, 16, 0, 6)
                local desc = lbl(row, cfg2.Description, 11,
                    theme.TextMuted, Enum.Font.Gotham, Enum.TextXAlignment.Left, 8)
                desc.Size     = UDim2.new(1, -72, 0, 14)
                desc.Position = UDim2.new(0, 16, 0, 24)
            else
                rName.Size     = UDim2.new(1, -72, 1, 0)
                rName.Position = UDim2.new(0, 16, 0, 0)
            end

            -- Toggle track
            local track = Instance.new("Frame")
            track.Size             = UDim2.new(0, 42, 0, 22)
            track.Position         = UDim2.new(1, -54, 0.5, -11)
            track.BackgroundColor3 = state and theme.Accent or theme.CardHover
            track.ZIndex           = 8
            track.Parent           = row
            corner(track, 11)

            local knob = Instance.new("Frame")
            knob.Size              = UDim2.new(0, 18, 0, 18)
            knob.Position          = state
                and UDim2.new(1, -20, 0.5, -9)
                or  UDim2.new(0,   2, 0.5, -9)
            knob.BackgroundColor3  = Color3.fromRGB(255, 255, 255)
            knob.ZIndex            = 9
            knob.Parent            = track
            corner(knob, 9)

            local function refresh()
                tw(track, 0.2, {BackgroundColor3 = state and theme.Accent or theme.CardHover})
                tw(knob,  0.2, {Position = state
                    and UDim2.new(1, -20, 0.5, -9)
                    or  UDim2.new(0,   2, 0.5, -9)})
                tw(rStroke, 0.2, {Color = state and theme.Accent or theme.Stroke})
            end

            row.MouseButton1Click:Connect(function()
                state = not state
                refresh()
                if cfg2.Callback then cfg2.Callback(state) end
            end)
            row.MouseEnter:Connect(function()
                tw(row, 0.12, {BackgroundColor3=theme.CardHover})
            end)
            row.MouseLeave:Connect(function()
                tw(row, 0.12, {BackgroundColor3=theme.Card})
            end)
            refresh()

            local Toggle = {}
            function Toggle:Set(v)
                state = v; refresh()
                if cfg2.Callback then cfg2.Callback(state) end
            end
            function Toggle:Get() return state end
            return Toggle
        end

        --------------------------------------------------------
        -- SLIDER
        --------------------------------------------------------
        function Tab:AddSlider(cfg2)
            cfg2 = cfg2 or {}
            local minV  = cfg2.Min     or 0
            local maxV  = cfg2.Max     or 100
            local suf   = cfg2.Suffix  or ""
            local value = math.clamp(cfg2.Default or minV, minV, maxV)

            local box = Instance.new("Frame")
            box.Size             = UDim2.new(1, 0, 0, 56)
            box.BackgroundColor3 = theme.Card
            box.LayoutOrder      = nextOrder()
            box.ZIndex           = 7
            box.Parent           = page
            corner(box, 10)
            mkStroke(box, theme.Stroke, 1)

            local sName = lbl(box, cfg2.Name or "Slider", 13,
                theme.Text, Enum.Font.GothamSemibold, Enum.TextXAlignment.Left, 8)
            sName.Size     = UDim2.new(1, -70, 0, 20)
            sName.Position = UDim2.new(0, 16, 0, 8)

            local valLbl = lbl(box, tostring(value)..suf, 12,
                theme.Accent, Enum.Font.GothamBold, Enum.TextXAlignment.Right, 8)
            valLbl.Size     = UDim2.new(0, 56, 0, 20)
            valLbl.Position = UDim2.new(1, -68, 0, 8)

            -- Track background
            local track = Instance.new("Frame")
            track.Size             = UDim2.new(1, -32, 0, 5)
            track.Position         = UDim2.new(0, 16, 0, 40)
            track.BackgroundColor3 = theme.CardHover
            track.ZIndex           = 8
            track.Parent           = box
            corner(track, 3)

            -- Fill
            local fill = Instance.new("Frame")
            fill.Size              = UDim2.new((value-minV)/(maxV-minV), 0, 1, 0)
            fill.BackgroundColor3  = theme.Accent
            fill.ZIndex            = 9
            fill.Parent            = track
            corner(fill, 3)

            local fillGrad = Instance.new("UIGradient")
            fillGrad.Color = ColorSequence.new(theme.AccentDim, theme.AccentGlow)
            fillGrad.Parent = fill

            -- Dot handle
            local dot2 = Instance.new("Frame")
            dot2.Size             = UDim2.new(0, 13, 0, 13)
            dot2.AnchorPoint      = Vector2.new(0.5, 0.5)
            dot2.Position         = UDim2.new((value-minV)/(maxV-minV), 0, 0.5, 0)
            dot2.BackgroundColor3 = Color3.fromRGB(255,255,255)
            dot2.ZIndex           = 10
            dot2.Parent           = track
            corner(dot2, 7)
            mkStroke(dot2, theme.Accent, 2)

            local isDragging = false

            local function setVal(x)
                local abs = track.AbsolutePosition.X
                local w   = track.AbsoluteSize.X
                local pct = math.clamp((x - abs) / w, 0, 1)
                if cfg2.Integer ~= false then
                    value = math.floor(minV + pct * (maxV - minV) + 0.5)
                else
                    value = minV + pct * (maxV - minV)
                    value = math.floor(value * 100 + 0.5) / 100
                end
                local p = (value - minV) / (maxV - minV)
                tw(fill,  0.04, {Size = UDim2.new(p, 0, 1, 0)})
                tw(dot2,  0.04, {Position = UDim2.new(p, 0, 0.5, 0)})
                valLbl.Text = tostring(value)..suf
                if cfg2.Callback then cfg2.Callback(value) end
            end

            track.InputBegan:Connect(function(inp)
                if inp.UserInputType == Enum.UserInputType.MouseButton1 then
                    isDragging = true; setVal(inp.Position.X)
                end
            end)
            UserInputService.InputChanged:Connect(function(inp)
                if isDragging and inp.UserInputType == Enum.UserInputType.MouseMovement then
                    setVal(inp.Position.X)
                end
            end)
            UserInputService.InputEnded:Connect(function(inp)
                if inp.UserInputType == Enum.UserInputType.MouseButton1 then
                    isDragging = false
                end
            end)

            local Slider = {}
            function Slider:Set(v)
                value = math.clamp(v, minV, maxV)
                local p = (value-minV)/(maxV-minV)
                tw(fill,  0.14, {Size = UDim2.new(p, 0, 1, 0)})
                tw(dot2,  0.14, {Position = UDim2.new(p, 0, 0.5, 0)})
                valLbl.Text = tostring(value)..suf
                if cfg2.Callback then cfg2.Callback(value) end
            end
            function Slider:Get() return value end
            return Slider
        end

        function Tab:AddDropdown(cfg2)
            cfg2 = cfg2 or {}
            local opts     = cfg2.Options or {}
            local selected = cfg2.Default or (opts[1] or "None")
            local dropOpen = false

            local box = Instance.new("Frame")
            box.Size              = UDim2.new(1, 0, 0, 44)
            box.BackgroundColor3  = theme.Card
            box.LayoutOrder       = nextOrder()
            box.ClipsDescendants  = false
            box.ZIndex            = 7
            box.Parent            = page
            corner(box, 10)
            local dStroke = mkStroke(box, theme.Stroke, 1)

            local dName = lbl(box, cfg2.Name or "Dropdown", 13,
                theme.Text, Enum.Font.GothamSemibold, Enum.TextXAlignment.Left, 8)
            dName.Size     = UDim2.new(0.46, 0, 1, 0)
            dName.Position = UDim2.new(0, 16, 0, 0)

            local selBtn = Instance.new("TextButton")
            selBtn.Size             = UDim2.new(0.47, 0, 0, 28)
            selBtn.Position         = UDim2.new(0.5, 0, 0.5, -14)
            selBtn.BackgroundColor3 = theme.CardHover
            selBtn.Text             = selected
            selBtn.TextColor3       = theme.Accent
            selBtn.Font             = Enum.Font.GothamSemibold
            selBtn.TextSize         = 12
            selBtn.AutoButtonColor  = false
            selBtn.ZIndex           = 8
            selBtn.Parent           = box
            corner(selBtn, 7)
            mkPad(selBtn, 0, 0, 8, 22)

            -- chevron inside button
            local chev = lbl(selBtn, "▾", 11, theme.TextMuted, Enum.Font.GothamBold,
                Enum.TextXAlignment.Right, 9)
            chev.Size     = UDim2.new(1, -4, 1, 0)
            chev.Position = UDim2.new(0, 0, 0, 0)

            -- Drop panel
            local dp = Instance.new("Frame")
            dp.Size              = UDim2.new(0.47, 0, 0, 0)
            dp.Position          = UDim2.new(0.5, 0, 1, 5)
            dp.BackgroundColor3  = theme.Panel
            dp.ClipsDescendants  = true
            dp.ZIndex            = 25
            dp.Visible           = false
            dp.Parent            = box
            corner(dp, 8)
            mkStroke(dp, theme.Accent, 1)
            mkPad(dp, 4, 4, 4, 4)

            local dpLayout = Instance.new("UIListLayout")
            dpLayout.Padding   = UDim.new(0, 2)
            dpLayout.SortOrder = Enum.SortOrder.LayoutOrder
            dpLayout.Parent    = dp

            local function closeD()
                dropOpen = false
                tw(dp, 0.15, {Size = UDim2.new(0.47, 0, 0, 0)})
                task.delay(0.16, function() dp.Visible = false end)
                tw(chev, 0.15, {Rotation = 0})
            end

            local function buildOptions()
                for _, c in pairs(dp:GetChildren()) do
                    if c:IsA("TextButton") then c:Destroy() end
                end
                for i, opt in ipairs(opts) do
                    local ob = Instance.new("TextButton")
                    ob.Size             = UDim2.new(1, 0, 0, 26)
                    ob.BackgroundColor3 = theme.Card
                    ob.Text             = opt
                    ob.TextColor3       = opt == selected and theme.Accent or theme.Text
                    ob.Font             = Enum.Font.Gotham
                    ob.TextSize         = 11
                    ob.AutoButtonColor  = false
                    ob.LayoutOrder      = i
                    ob.ZIndex           = 26
                    ob.Parent           = dp
                    corner(ob, 5)
                    ob.MouseEnter:Connect(function()
                        tw(ob, 0.1, {BackgroundColor3=theme.CardHover, TextColor3=theme.AccentGlow})
                    end)
                    ob.MouseLeave:Connect(function()
                        tw(ob, 0.1, {BackgroundColor3=theme.Card,
                            TextColor3 = opt == selected and theme.Accent or theme.Text})
                    end)
                    ob.MouseButton1Click:Connect(function()
                        selected     = opt
                        selBtn.Text  = opt
                        closeD()
                        buildOptions()
                        if cfg2.Callback then cfg2.Callback(opt) end
                    end)
                end
            end
            buildOptions()

            selBtn.MouseButton1Click:Connect(function()
                dropOpen = not dropOpen
                if dropOpen then
                    dp.Visible = true
                    local h = math.min(#opts * 28 + 10, 160)
                    tw(dp, 0.18, {Size = UDim2.new(0.47, 0, 0, h)})
                    tw(chev, 0.18, {Rotation = 180})
                    tw(dStroke, 0.18, {Color = theme.Accent})
                else
                    closeD()
                    tw(dStroke, 0.18, {Color = theme.Stroke})
                end
            end)

            local DD = {}
            function DD:Set(v) selected=v; selBtn.Text=v; buildOptions() end
            function DD:Get() return selected end
            function DD:SetOptions(newOpts) opts=newOpts; buildOptions() end
            return DD
        end
