local function hasFn(name)
	return type(_G[name]) == "function" or type(getgenv and getgenv()[name]) == "function" or type(_G[name]) == "function"
end

local function safeFn(name, ...)
	local fn = rawget(_G, name)
	if type(fn) == "function" then
		return pcall(fn, ...)
	end
	if type(_G[name]) == "function" then
		return pcall(_G[name], ...)
	end
	return false
end

local function canUseFileFns()
	return type(writefile) == "function"
		and type(readfile) == "function"
		and type(isfile) == "function"
		and type(makefolder) == "function"
end

local function folderExists(path)
	if type(isfolder) == "function" then
		local ok, result = pcall(isfolder, path)
		return ok and result
	end
	return false
end

local function ensureFolder(path)
	if canUseFileFns() then
		if not folderExists(path) then
			pcall(makefolder, path)
		end
	end
end

local function safeWrite(path, data)
	if canUseFileFns() then
		ensureFolder(AnimeUI.ConfigFolder)
		return pcall(writefile, path, data)
	end
	return false
end

local function safeRead(path)
	if canUseFileFns() and isfile(path) then
		local ok, result = pcall(readfile, path)
		if ok then
			return result
		end
	end
	return nil
end

local function sanitizeFileName(name)
	name = tostring(name or "Default")
	name = name:gsub("[\\/:*?\"<>|]", "_")
	return name
end

local function makeCorner(parent, radius)
	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, radius or 10)
	corner.Parent = parent
	return corner
end

local function makeStroke(parent, color, thickness)
	local stroke = Instance.new("UIStroke")
	stroke.Color = color
	stroke.Thickness = thickness or 1
	stroke.Parent = parent
	return stroke
end

local function makePadding(parent, left, right, top, bottom)
	local pad = Instance.new("UIPadding")
	pad.PaddingLeft = UDim.new(0, left or 0)
	pad.PaddingRight = UDim.new(0, right or 0)
	pad.PaddingTop = UDim.new(0, top or 0)
	pad.PaddingBottom = UDim.new(0, bottom or 0)
	pad.Parent = parent
	return pad
end

local function tween(obj, time, props)
	local t = TweenService:Create(
		obj,
		TweenInfo.new(time or 0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
		props
	)
	t:Play()
	return t
end

local function clamp(v, min, max)
	if v < min then
		return min
	elseif v > max then
		return max
	end
	return v
end

local function createTextLabel(parent, text, size, xAlign, color, font, textSize)
	local label = Instance.new("TextLabel")
	label.BackgroundTransparency = 1
	label.Text = text or ""
	label.TextColor3 = color or Color3.new(1, 1, 1)
	label.Font = font or Enum.Font.Gotham
	label.TextSize = textSize or 14
	label.TextWrapped = false
	label.TextXAlignment = xAlign or Enum.TextXAlignment.Left
	label.Size = size or UDim2.new(1, 0, 0, 20)
	label.Parent = parent
	return label
end

function AnimeUI:_resolveTheme(theme)
	if typeof(theme) == "table" then
		return theme
	end
	return self.Themes[theme or "Sakura"] or self.Themes.Sakura
end

	function window:_registerFlag(flag, defaultValue, setter)
		if not flag or flag == "" then
			return defaultValue
		end

		if self._flags[flag] == nil then
			self._flags[flag] = defaultValue
		end

		self._setters[flag] = setter

		if setter then
			pcall(setter, self._flags[flag])
		end

		return self._flags[flag]
	end

	function window:_setFlag(flag, value)
		if not flag or flag == "" then
			return
		end
		self._flags[flag] = value
		local setter = self._setters[flag]
		if setter then
			pcall(setter, value)
		end
		if self._autoSave and self._defaultConfigName then
			self:SaveConfig(self._defaultConfigName)
		end
	end

	function window:SetBackground(imageId)
		self.BackgroundImage.Image = imageId or ""
	end

	function window:SetIcon(imageId)
		self.Icon.Image = imageId or ""
	end

	function window:SetTheme(themeName)
		local newTheme = self:_resolveTheme(themeName)
		if not newTheme then
			return
		end

		self.Theme = newTheme
		self.Main.BackgroundColor3 = newTheme.Background
		self.BackgroundImage.ImageTransparency = config.BackgroundTransparency or 0.78
		self.Main.UIStroke.Color = newTheme.Accent
		self.Gui.Main = self.Main
		topbar.BackgroundColor3 = newTheme.Panel
		title.TextColor3 = newTheme.Text
		sidebar.BackgroundColor3 = newTheme.Panel

		for _, btn in ipairs(self._tabButtons) do
			btn.BackgroundColor3 = newTheme.Panel2
			btn.TextColor3 = newTheme.Text
		end
	end

	function window:SaveConfig(name)
		local fileName = sanitizeFileName(name or self._defaultConfigName)
		local path = AnimeUI.ConfigFolder .. "/" .. fileName .. ".json"

		if not canUseFileFns() then
			return false, "File APIs not available"
		end

		ensureFolder(AnimeUI.ConfigFolder)

		local encoded = HttpService:JSONEncode(self._flags)
		local ok = safeWrite(path, encoded)
		return ok
	end

	function window:LoadConfig(name)
		local fileName = sanitizeFileName(name or self._defaultConfigName)
		local path = AnimeUI.ConfigFolder .. "/" .. fileName .. ".json"

		if not canUseFileFns() then
			return false, "File APIs not available"
		end

		local raw = safeRead(path)
		if not raw or raw == "" then
			return false, "Config not found"
		end

		local ok, decoded = pcall(function()
			return HttpService:JSONDecode(raw)
		end)
		if not ok or type(decoded) ~= "table" then
			return false, "Invalid config"
		end

		for flag, value in pairs(decoded) do
			self:_setFlag(flag, value)
		end

		return true
	end

		function tab:AddToggle(cfg)
			cfg = cfg or {}

			local flag = cfg.Flag
			local state = cfg.CurrentValue
			if state == nil then
				state = cfg.Default or false
			end

			local holder = Instance.new("Frame")
			holder.Size = UDim2.new(1, 0, 0, 44)
			holder.BackgroundColor3 = theme.Panel2
			holder.BorderSizePixel = 0
			holder.Parent = page
			makeCorner(holder, 10)

			local label = createTextLabel(holder, cfg.Name or "Toggle", UDim2.new(1, -70, 1, 0), Enum.TextXAlignment.Left, theme.Text, Enum.Font.Gotham, 14)
			label.Position = UDim2.new(0, 12, 0, 0)

			local switch = Instance.new("TextButton")
			switch.Size = UDim2.new(0, 48, 0, 24)
			switch.Position = UDim2.new(1, -58, 0.5, -12)
			switch.BackgroundColor3 = state and theme.Accent or Color3.fromRGB(90, 90, 90)
			switch.Text = ""
			switch.BorderSizePixel = 0
			switch.AutoButtonColor = false
			switch.Parent = holder
			makeCorner(switch, 12)

			local function apply(v)
				state = not not v
				switch.BackgroundColor3 = state and theme.Accent or Color3.fromRGB(90, 90, 90)
			end

			if flag then
				state = self:_registerFlag(flag, state, apply)
			else
				apply(state)
			end

			switch.MouseButton1Click:Connect(function()
				state = not state
				apply(state)
				if flag then
					self:_setFlag(flag, state)
				end
				if type(cfg.Callback) == "function" then
					cfg.Callback(state)
				end
			end)

			return {
				Set = function(_, v)
					state = not not v
					apply(state)
					if flag then
						self:_setFlag(flag, state)
					end
				end,
				Get = function()
					return state
				end
			}
		end

		function tab:AddSlider(cfg)
			cfg = cfg or {}

			local flag = cfg.Flag
			local min = tonumber(cfg.Min) or 0
			local max = tonumber(cfg.Max) or 100
			local decimal = cfg.Decimals or 0

			local value = tonumber(cfg.CurrentValue)
			if value == nil then
				value = tonumber(cfg.Default) or min
			end

			local holder = Instance.new("Frame")
			holder.Size = UDim2.new(1, 0, 0, 62)
			holder.BackgroundColor3 = theme.Panel2
			holder.BorderSizePixel = 0
			holder.Parent = page
			makeCorner(holder, 10)

			local label = createTextLabel(holder, cfg.Name or "Slider", UDim2.new(1, -16, 0, 20), Enum.TextXAlignment.Left, theme.Text, Enum.Font.Gotham, 14)
			label.Position = UDim2.new(0, 12, 0, 6)

			local valueLabel = createTextLabel(holder, tostring(value), UDim2.new(0, 80, 0, 20), Enum.TextXAlignment.Right, theme.Muted, Enum.Font.Gotham, 13)
			valueLabel.Position = UDim2.new(1, -92, 0, 6)

			local bar = Instance.new("Frame")
			bar.Size = UDim2.new(1, -24, 0, 8)
			bar.Position = UDim2.new(0, 12, 0, 38)
			bar.BackgroundColor3 = Color3.fromRGB(80, 80, 90)
			bar.BorderSizePixel = 0
			bar.Parent = holder
			makeCorner(bar, 8)

			local fill = Instance.new("Frame")
			fill.Size = UDim2.new(0, 0, 1, 0)
			fill.BackgroundColor3 = theme.Accent
			fill.BorderSizePixel = 0
			fill.Parent = bar
			makeCorner(fill, 8)

			local dragging = false

			local function round(num, decimals)
				local mult = 10 ^ (decimals or 0)
				return math.floor(num * mult + 0.5) / mult
			end

			local function apply(v)
				value = clamp(tonumber(v) or min, min, max)
				value = round(value, decimal)
				local pct = (value - min) / math.max((max - min), 0.0001)
				fill.Size = UDim2.new(pct, 0, 1, 0)
				valueLabel.Text = tostring(value)
			end

			if flag then
				value = tonumber(self:_registerFlag(flag, value, apply)) or value
			else
				apply(value)
			end

			local function setFromX(x)
				local pct = clamp((x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
				local newValue = min + ((max - min) * pct)
				apply(newValue)
				if flag then
					self:_setFlag(flag, value)
				end
				if type(cfg.Callback) == "function" then
					cfg.Callback(value)
				end
			end

			bar.InputBegan:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1 then
					dragging = true
					setFromX(input.Position.X)
				end
			end)

			UserInputService.InputChanged:Connect(function(input)
				if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
					setFromX(input.Position.X)
				end
			end)

			UserInputService.InputEnded:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1 then
					dragging = false
				end
			end)

			return {
				Set = function(_, v)
					apply(v)
					if flag then
						self:_setFlag(flag, value)
					end
				end,
				Get = function()
					return value
				end
			}
		end

		function tab:AddDropdown(cfg)
			cfg = cfg or {}

			local flag = cfg.Flag
			local options = cfg.Options or {}
			local selected = cfg.CurrentValue or cfg.Default or options[1] or "None"
			local open = false

			local holder = Instance.new("Frame")
			holder.Size = UDim2.new(1, 0, 0, 44)
			holder.BackgroundColor3 = theme.Panel2
			holder.BorderSizePixel = 0
			holder.ClipsDescendants = true
			holder.Parent = page
			makeCorner(holder, 10)

			local label = createTextLabel(holder, cfg.Name or "Dropdown", UDim2.new(1, -80, 0, 20), Enum.TextXAlignment.Left, theme.Text, Enum.Font.Gotham, 14)
			label.Position = UDim2.new(0, 12, 0, 6)

			local button = Instance.new("TextButton")
			button.Size = UDim2.new(1, -24, 0, 26)
			button.Position = UDim2.new(0, 12, 0, 20)
			button.BackgroundColor3 = Color3.fromRGB(40, 40, 50)
			button.BorderSizePixel = 0
			button.TextColor3 = theme.Muted
			button.Font = Enum.Font.Gotham
			button.TextSize = 13
			button.AutoButtonColor = false
			button.Text = tostring(selected)
			button.Parent = holder
			makeCorner(button, 8)

			local list = Instance.new("Frame")
			list.Size = UDim2.new(1, -24, 0, 0)
			list.Position = UDim2.new(0, 12, 0, 48)
			list.BackgroundTransparency = 1
			list.BorderSizePixel = 0
			list.Parent = holder

			local listLayout = Instance.new("UIListLayout")
			listLayout.Padding = UDim.new(0, 6)
			listLayout.SortOrder = Enum.SortOrder.LayoutOrder
			listLayout.Parent = list

			local function apply(v)
				selected = v
				button.Text = tostring(v)
			end

			if flag then
				selected = self:_registerFlag(flag, selected, apply)
			else
				apply(selected)
			end

			local function rebuild()
				for _, child in ipairs(list:GetChildren()) do
					if child:IsA("TextButton") then
						child:Destroy()
					end
				end

				for _, opt in ipairs(options) do
					local optButton = Instance.new("TextButton")
					optButton.Size = UDim2.new(1, 0, 0, 28)
					optButton.BackgroundColor3 = Color3.fromRGB(40, 40, 50)
					optButton.BorderSizePixel = 0
					optButton.Text = tostring(opt)
					optButton.TextColor3 = theme.Text
					optButton.Font = Enum.Font.Gotham
					optButton.TextSize = 13
					optButton.AutoButtonColor = false
					optButton.Parent = list
					makeCorner(optButton, 8)

					optButton.MouseButton1Click:Connect(function()
						apply(opt)
						if flag then
							self:_setFlag(flag, selected)
						end
						if type(cfg.Callback) == "function" then
							cfg.Callback(selected)
						end
						open = false
						holder.Size = UDim2.new(1, 0, 0, 44)
					end)
				end

				list.Size = UDim2.new(1, -24, 0, (#options * 34) + math.max(0, (#options - 1) * 6))
			end

			rebuild()

			button.MouseButton1Click:Connect(function()
				open = not open
				holder.Size = open and UDim2.new(1, 0, 0, 52 + ((#options * 34) + math.max(0, (#options - 1) * 6))) or UDim2.new(1, 0, 0, 44)
			end)

			return {
				Set = function(_, v)
					apply(v)
					if flag then
						self:_setFlag(flag, selected)
					end
				end,
				Get = function()
					return selected
				end,
				Refresh = function(_, newOptions)
					options = newOptions or options
					rebuild()
				end
			}
		end

		return tab
	end

	return window
end

return AnimeUI
